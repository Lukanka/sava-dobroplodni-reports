// 1. ИЗВЛИЧАНЕ НА ВСИЧКИ ВЪЗМОЖНИ ДОСТАВЧИЦИ И ОБЕКТИ
const allInvoices = dv.pages().filter(p => p.фирма && p.дата_фактура);

// Помощна функция за пълно изчистване на Obsidian линкове и пътища до просто име
function cleanName(rawVal) {
    if (!rawVal) return "";
    let clean = rawVal.toString();
    if (clean.includes("|")) {
        clean = clean.split("|").pop();
    }
    return clean.replace(/[\[\]]/g, "").replace(/\.md$/, "").trim();
}

// Вземаме уникалните доставчици от фактурите
const uniqueSuppliers = [...new Set(allInvoices.map(p => cleanName(p.фирма)))].filter(Boolean).sort();

// Вземаме абсолютно ВСИЧКИ обекти директно от папката с обектите ти
const uniqueObjects = dv.pages('"2_Areas/01_Строителни обекти"')
    .map(p => cleanName(p.file.name))
    .filter(Boolean)
    .sort();

// 2. СЪЗДАВАНЕ НА ЕЛЕГАНТЕН, КОМПАКТЕН ИНТЕРФЕЙС С КАЛЕНДАРИ
const container = dv.container.createEl("div", { 
    cls: "filter-container", 
    attr: { 
        style: "display: flex; gap: 15px; margin-bottom: 22px; align-items: center; background: #1a1a1a; padding: 10px 18px; border-radius: 6px; border: 1px solid #2e2e2e; box-shadow: 0 4px 6px rgba(0,0,0,0.15); max-width: fit-content; flex-wrap: wrap;" 
    } 
});

const selectStyle = "background: #262626; color: #e5e5e5; border: 1px solid #3f3f3f; padding: 4px 10px; border-radius: 4px; cursor: pointer; font-size: 0.9em; max-width: 250px; text-overflow: ellipsis; outline: none;";

// Поправеният стил на календара: добавен padding-left за иконката и коригирано двоеточие
const dateStyle = "background: #262626; color: #e5e5e5; border: 1px solid #3f3f3f; padding: 4px 8px 4px 28px; border-radius: 4px; cursor: pointer; font-size: 0.9em; width: 155px; outline: none; font-family: inherit;";

// Падащо меню за Доставчик
container.createEl("span", { text: "Доставчик: ", attr: { style: "color: #a3a3a3; font-weight: 600; font-size: 0.9em;" } });
const supplierSelect = container.createEl("select", { attr: { style: selectStyle } });
supplierSelect.createEl("option", { text: "— Всички —", value: "" });
uniqueSuppliers.forEach(s => supplierSelect.createEl("option", { text: s, value: s }));

// Падащо меню за Обект
container.createEl("span", { text: "Обект: ", attr: { style: "color: #a3a3a3; font-weight: 600; font-size: 0.9em; margin-left: 5px;" } });
const objectSelect = container.createEl("select", { attr: { style: selectStyle } });
objectSelect.createEl("option", { text: "— Всички —", value: "" });
uniqueObjects.forEach(o => objectSelect.createEl("option", { text: o, value: o }));

// Календар за Начална дата
container.createEl("span", { text: "От: ", attr: { style: "color: #a3a3a3; font-weight: 600; font-size: 0.9em; margin-left: 5px;" } });
const startDateInput = container.createEl("input", { attr: { type: "date", style: dateStyle } });

// Календар за Крайна дата
container.createEl("span", { text: "До: ", attr: { style: "color: #a3a3a3; font-weight: 600; font-size: 0.9em; margin-left: 5px;" } });
const endDateInput = container.createEl("input", { attr: { type: "date", style: dateStyle } });

// Бутон за бързо изчистване на датите
const resetDatesBtn = container.createEl("button", { 
    text: "✕", 
    attr: { 
        style: "background: #3a3a3a; color: #ccc; border: none; padding: 2px 7px; border-radius: 4px; cursor: pointer; font-size: 0.8em; font-weight: bold;",
        title: "Изчисти филтрите за дата"
    } 
});

// Добавяне на hover ефекти на полетата
[supplierSelect, objectSelect, startDateInput, endDateInput, resetDatesBtn].forEach(el => {
    el.addEventListener("mouseenter", () => el.style.borderColor = "#525252");
    el.addEventListener("mouseleave", () => el.style.borderColor = "#3f3f3f");
});

// Контейнер за таблицата
const resultsArea = dv.container.createEl("div");

// 3. ФУНКЦИЯ ЗА ФИЛТРИРАНЕ И ИЗРИСУВАНЕ НА ТАБЛИЦАТА
function updateTable() {
    resultsArea.innerHTML = ""; // Изчистваме старите данни
    
    const selectedSupplier = supplierSelect.value.toLowerCase();
    const selectedObject = objectSelect.value.toLowerCase();
    
    // Вземаме датите от новите календарни полета
    const startDateVal = startDateInput.value; // формат YYYY-MM-DD
    const endDateVal = endDateInput.value;     // формат YYYY-MM-DD
    
    const startDate = startDateVal ? moment(startDateVal) : null;
    const endDate = endDateVal ? moment(endDateVal) : null;
    
    // Филтрираме
    const filtered = Array.from(allInvoices).filter(p => {
        const cleanInvoiceSupplier = cleanName(p.фирма).toLowerCase();
        const cleanInvoiceObject = p.обект ? cleanName(p.обект).toLowerCase() : "";

        const matchesSupplier = selectedSupplier ? cleanInvoiceSupplier === selectedSupplier : true;
        const matchesObject = selectedObject ? cleanInvoiceObject === selectedObject : true;
        
        // Проверка на датите
        const rawDate = p.дата_фактура;
        const invoiceDate = moment(rawDate.toString(), ["DD.MM.YYYY", "YYYY-MM-DD"]);
        const afterStart = startDate && invoiceDate.isValid() ? invoiceDate.isSameOrAfter(startDate, 'day') : true;
        const beforeEnd = endDate && invoiceDate.isValid() ? invoiceDate.isSameOrBefore(endDate, 'day') : true;
        
        return matchesSupplier && matchesObject && afterStart && beforeEnd;
    });

    // Сортираме по дата (най-новите отгоре)
    filtered.sort((a, b) => {
        const dateA = moment(a.дата_фактура.toString(), ["DD.MM.YYYY", "YYYY-MM-DD"]).valueOf() || 0;
        const dateB = moment(b.дата_фактура.toString(), ["DD.MM.YYYY", "YYYY-MM-DD"]).valueOf() || 0;
        return dateB - dateA;
    });

    let totalAmount = 0;
    filtered.forEach(p => totalAmount += (parseFloat(p.данъчна_основа_евро) || 0));

    // Показване на резултата
    if (filtered.length > 0) {
        resultsArea.createEl("h3", { 
            text: `💰 Обща данъчна основа: ${totalAmount.toLocaleString('bg-BG', { minimumFractionDigits: 2, maximumFractionDigits: 2 })} €`, 
            attr: { style: "margin-top: 15px; font-weight: 600; color: #f5f5f5;" } 
        });
        
        // СЪЗДАВАНЕ НА СТАБИЛНА ТАБЛИЦА
        const table = resultsArea.createEl("table", { cls: "dataview table-view-table", attr: { style: "width: 100%; margin-top: 15px;" } });
        
        // Хедър на таблицата
        const thead = table.createEl("thead", { cls: "table-view-thead" });
        const trHead = thead.createEl("tr");
        ["Дата", "Файл (Фактура)", "Доставчик", "Обект", "Данъчна основа (без ДДС)"].forEach(h => {
            trHead.createEl("th", { text: h, attr: { style: "font-weight: bold; border-bottom: 1px solid #3f3f3f; padding: 8px 5px; text-align: left;" } });
        });

        // Тяло на таблицата
        const tbody = table.createEl("tbody", { cls: "table-view-tbody" });
        
        for (let p of filtered) {
            const rawDate = p.дата_фактура;
            const invoiceDate = moment(rawDate.toString(), ["DD.MM.YYYY", "YYYY-MM-DD"]);
            const formattedDate = invoiceDate.isValid() ? invoiceDate.format("DD.MM.YYYY") : rawDate;
            const cleanSum = parseFloat(p?.данъчна_основа_евро) || 0;

            const tr = tbody.createEl("tr");
            tr.createEl("td", { text: formattedDate, attr: { style: "padding: 8px 5px;" } });
            
            // Кликаем линк към фактурата
            const tdLink = tr.createEl("td", { attr: { style: "padding: 8px 5px;" } });
            const link = tdLink.createEl("a", { 
                text: p.file.name, 
                href: p.file.path, 
                cls: "internal-link" 
            });
            link.setAttribute("data-href", p.file.path); 
            
            tr.createEl("td", { text: cleanName(p.фирма), attr: { style: "padding: 8px 5px;" } });
            tr.createEl("td", { text: p.обект ? cleanName(p.обект) : "—", attr: { style: "padding: 8px 5px;" } });
            tr.createEl("td", { text: `${cleanSum.toLocaleString('bg-BG', { minimumFractionDigits: 2, maximumFractionDigits: 2 })} €`, attr: { style: "padding: 8px 5px;" } });
        }
    } else {
        resultsArea.createEl("p", { 
            text: "ℹ️ Няма намерени фактури за избраните критерии и дати.", 
            attr: { style: "color: #737373; font-style: italic; margin-top: 15px; font-size: 0.95em;" } 
        });
    }
}

// 4. СЛЕДЕНЕ ЗА ПРОМЕНИ (EVENTS)
supplierSelect.addEventListener("change", updateTable);
objectSelect.addEventListener("change", updateTable);
startDateInput.addEventListener("change", updateTable);
endDateInput.addEventListener("change", updateTable);

resetDatesBtn.addEventListener("click", () => {
    startDateInput.value = "";
    endDateInput.value = "";
    updateTable();
});

// Първоначално зареждане
updateTable();
